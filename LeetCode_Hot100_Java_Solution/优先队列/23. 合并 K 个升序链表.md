/**
 * Definition for singly-linked list.
 * public class ListNode {
 *     int val;
 *     ListNode next;
 *     ListNode() {}
 *     ListNode(int val) { this.val = val; }
 *     ListNode(int val, ListNode next) { this.val = val; this.next = next; }
 * }
 */
class Solution {
    // 最小のヒープを作成し、リンクリストのすべてのヘッドノードを格納し、一度にリンクリストから1つのノードを取り出し、このノードの次のノードを最小のヒープに入れます
    // これにより、一度に少なくとも現在のk個のノードが保証されます
    public ListNode mergeKLists(ListNode[] lists) {
        PriorityQueue<ListNode> pq = new PriorityQueue<>((a,b) -> a.val - b.val);
        ListNode dummy = new ListNode(-1);
        ListNode cur = dummy;
        for (ListNode node :lists) {
            if (node != null) { // 空であるかどうかを判断する必要があります。そうしないと、空のポインタ例外が発生し、空の値でプライオリティキューを追加することはできません。
                pq.offer(node);
            }  
        }
        while (!pq.isEmpty()) {
            cur.next = pq.poll();
            cur = cur.next;
            if (cur.next != null) {
                pq.offer(cur.next);
            }
        }
        return dummy.next;
    }
}
