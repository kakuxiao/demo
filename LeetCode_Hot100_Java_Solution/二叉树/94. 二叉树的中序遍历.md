/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
 // 2つの方法： 1.再帰2.反復
class Solution {
    List<Integer> res;
    public List<Integer> inorderTraversal(TreeNode root) {
       res = new ArrayList<>();
       Stack<TreeNode> stack = new Stack();
       TreeNode cur = root;
       while (!stack.isEmpty() || cur != null) {
            while (cur != null) {
                stack.push(cur);
                cur = cur.left;  // 左
            }
            cur = stack.pop();
            res.add(cur.val); // 中
            cur = cur.right; // 右
       }
       return res;
    }
   
      
    

}
