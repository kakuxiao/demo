class Solution {
    public int numTrees(int n) {
        int[] dp = new int[n+1]; //dp[i]n個のノードで構成されるバイナリ検索の数を表します
        dp[1] = 1;
        dp[0] = 1;
        //再帰式：いずれかの点iがルートノードとして選択されている場合、左のサブツリーは[1, i -1]であり、右のサブツリーは[i +1, n]であり、
        //dp[i] = ∑(dp[i-1] * dp[n-i])
        // dp[1] = 1
        // トラバーサルオーダー：前面から背面へ
        for (int i = 2; i <=n; i++) { // 合計ノード数
            for (int j = 1; j <= i; j++) { // ルートノードを実行するものを選択します
                dp[i] += dp[j-1]*dp[i-j];
            }
        }
       
        return dp[n]; 
    }
}
