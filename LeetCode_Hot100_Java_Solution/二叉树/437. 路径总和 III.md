/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    //targetSym: Depth - first searchへのすべてのパスとパスを計算します
    //関数rootSumを定義して、ノードから始まるパスの数とtargetSumから始まる記事の数を計算します。
    //pathSum関数は自身を再帰的に呼び出し、すべてのノードを通過します。
    // 時間複雑度O (N ^ 2 )、空間複雑度(N)
    // 接頭辞の和、時間の複雑さO (N )、空間の複雑さO (N)
    // ハッシュマップを使用して、シーケンシャルトラバーサルによるすべてのパスの合計を記録します。現在のノードをトラバーサルする場合、ハッシュマップにはパスと- targetsumがあります
    // このパスにノードから現在のノードへのパスがあり、ターゲットであることを示します
    // 最後に、他のノードが影響を受けるのを防ぐためにバックトラッキングが必要です。
    public int pathSum(TreeNode root, long targetSum) { // taegetSum オーバーフローします。longに変更する必要があります
       Map<Long, Integer> prefix = new HashMap<>();
       prefix.put(0L, 1); // パスと特殊ケースをtargetSumと正確に等しく扱うため
       return dfs(root, prefix, 0, targetSum);
    }
    public int dfs(TreeNode node, Map<Long, Integer> prefix, long curr,long targetSum) {
       if (node == null) {
        return 0;
       }
       int count = 0;
       curr += node.val;
       count = prefix.getOrDefault(curr - targetSum, 0);
       prefix.put(curr, prefix.getOrDefault(curr, 0) + 1);
       count += dfs(node.left, prefix, curr, targetSum);
       count += dfs(node.right, prefix, curr, targetSum);
       prefix.put(curr, prefix.get(curr) - 1); // 再帰的なリターンの後、テクノロジーを減らし、他のパスに影響を与える検索を防ぎます
       return count;
    }
}
