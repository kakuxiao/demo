/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode(int x) { val = x; }
 * }
 */
class Solution {
    // 再帰
    // 左側のサブツリーにノードpまたはqが含まれているかどうかを再帰的に確認します。
    // 右のサブツリーにノードpまたはqが含まれているかどうかを再帰的に確認します。
    // 現在のノード自体がｐまたはｑのうちの１つであり、その左または右のサブツリーが別のノードを含む場合、そのノードは最も近い共通の祖先である。
    // 現在のノードの左サブツリー及び右サブツリーがそれぞれターゲットノード（ ｐ及びｑ ）を含む場合、現在のノードは、最も近い共通の祖先である。
    // 上記のいずれにも当てはまらない場合は、ターゲットノードを含む見つかったサブツリーを返します（これは空の場合があり、現在のサブツリーにターゲットノードが含まれていないことを示します）。
    public TreeNode lowestCommonAncestor(TreeNode root, TreeNode p, TreeNode q) {
        if(root == null || root == p || root == q) {
            return root;
        }
        TreeNode left = lowestCommonAncestor(root.left, p, q);
        TreeNode right = lowestCommonAncestor(root.right, p, q);
        if (left!= null && right != null) {//いずれも空ではなく、共通の祖先がルートノードにあることを示します
            return root;
        }else if (left != null && right == null) { //片側が空で、片側が空でない場合は、パブリック祖先が空でない側であることを意味します
            return left;
        }else if (left == null && right != null) {
            return right;
        }
        return null;
    }
}
