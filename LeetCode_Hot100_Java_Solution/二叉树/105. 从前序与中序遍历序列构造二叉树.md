/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
public class Solution {
    // バイナリツリーの場合、プリアンブルトラバーサルの第１の値はルートノードの値であり、左右のサブツリーは、ルートノードの値及び中間順位トラバーサルに基づいて取得することができる。
    // 左右のサブツリーのルートノードを再帰的に取得する
    // シーケンシャルトラバーサルの最初の要素を取得して、ルートノードを作成します。
    // 中間順序トラバーサルのルートノードの位置を見つけます。これは、中間順序トラバーサルを左側のツリーの左側のサブツリーと右側のツリーの右側のサブツリーの2つの部分に分割します。
    // 左側のサブツリーの順序と中央値のトラバーサル結果を使用して、左側のサブツリーを再帰的に再構築します。
    // 右のサブツリーの順序と中央値のトラバーサル結果を使用して、右のサブツリーを再帰的に再構築します。
    // ルートノードに戻る
   int preIndex = 0;
    HashMap<Integer, Integer> hs = new HashMap<>();
    public TreeNode buildTree(int[] preorder, int[] inorder) {
        for (int i = 0; i < inorder.length; i++) {
            hs.put(inorder[i], i);
        }
        return buildTreeRecursive(preorder, 0, inorder.length - 1);    
    }
    // recursive 再帰
    // プリアンブルトラバーサルの順序でサブツリーを作成し、左側のサブツリーと右側のサブツリーの範囲は、中間順序のトラバーサルによって取得されます。
    public TreeNode buildTreeRecursive(int[] preorder, int left , int right) {
        if (left > right) {
            return null;
        }
        int rootVal = preorder[preIndex++]; // 前のシーケンスで横断された最初のノードは、常に現在のサブツリーのルートです
        TreeNode root = new TreeNode(rootVal);
        int index = hs.get(rootVal);
        root.left = buildTreeRecursive(preorder, left, index - 1);
        root.right = buildTreeRecursive(preorder, index + 1, right);
        return root;
    }
   
}
