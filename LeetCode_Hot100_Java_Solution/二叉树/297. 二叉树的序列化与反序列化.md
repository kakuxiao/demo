/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode(int x) { val = x; }
 * }
 */
// バイナリツリーを文字列にシリアライズし、元のツリー構造にデシリアライズすることができます
// シリアル化、シーケンシャルトラバーサル。デシリアライズし、最初と最後の"[]"を削除し、文字列を","で分割し、シーケンシャルトラバーサルによってツリーを復元します
// シーケンストラバーサル
public class Codec {
  
    // シリアライゼーション、ツリーのシーケンシャルトラバーサル。注：最後に「、」を削除してください
    public String serialize(TreeNode root) {
        if (root == null) {
            return "[]";
        }
        Queue<TreeNode> que = new LinkedList<>();
        StringBuffer sb = new StringBuffer();
        sb.append("[");
        que.offer(root);
        while (!que.isEmpty()) {
            TreeNode tmp = que.poll();
            if (tmp != null) {
                que.offer(tmp.left);
                que.offer(tmp.right); // ここでは、最終結果はさらにいくつかのヌルになり、シリアライズされたバイナリツリーには影響しません
                sb.append(tmp.val + ",");
            } else {
                sb.append("null,"); 
            }
        }
        sb.deleteCharAt(sb.length() - 1);
        sb.append("]");
        return sb.toString();
    }

    // Decodes your encoded data to tree.
    // 文字列を文字列の配列に分割する
    // 文字列をInteger Integer.parseInt ()に変換するメソッド;
    public TreeNode deserialize(String data) {
       if (data.equals("[]")) {
            return null;
       }
       String[] val = data.substring(1, data.length() - 1).split(",");
       TreeNode root = new TreeNode(Integer.parseInt(val[0]));
       Queue<TreeNode> que = new LinkedList<>();
       que.offer(root);
       int i = 1;
       while (!que.isEmpty()) {
            TreeNode tmp = que.poll();
            if (!val[i].equals("null")) {
                tmp.left = new TreeNode(Integer.parseInt(val[i]));
                que.offer(tmp.left);
            } 
            i++;
            if (!val[i].equals("null")) {
                tmp.right = new TreeNode(Integer.parseInt(val[i]));
                que.offer(tmp.right);
            }
            i++;
        }
        return root;
    }
}
