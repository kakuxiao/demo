   // 階層的なトラバーサル、古典的なアルゴリズムの問題、最初にヘッドノードをキューに追加し、次にキュー内の要素を順番にポーリング（ ）し、左右のノードが空でない場合は、左右のノードをキューに追加します
    public List<List<Integer>> levelOrder(TreeNode root) {
         List<List<Integer>> res = new ArrayList<>();
         if (root == null) {
            return res;
         }
         Queue<TreeNode> que = new LinkedList<>();
         que.offer(root);
         while (!que.isEmpty()) {
            int len = que.size();
            List<Integer> level = new ArrayList<>();
            while (len > 0) {
                TreeNode tmp = que.poll();
                level.add(tmp.val);
                if (tmp.left != null) {
                    que.offer(tmp.left);
                }
                if (tmp.right != null) {
                    que.offer(tmp.right);
                }
                len--;
            }
            res.add(level);
         }
         return res;
    }
    
}
