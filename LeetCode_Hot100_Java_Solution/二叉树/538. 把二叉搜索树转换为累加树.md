/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    // このソリューションは、再帰的なアプローチを使用します。アルゴリズムの主なステップは次のとおりです。
    // 現在のノードが空の場合、nullを返します。
    // まず、現在のノードよりも大きいノードを最初に処理する必要があるため、右のサブツリーは再帰的に処理されます。
    // 右のサブツリーを処理した後、現在のノードの値をSUM変数に累積し、現在のノードの値をSUMに更新します。
    // 最後に、左側のサブツリーを再帰的に処理します。
    int sum = 0;
    public TreeNode convertBST(TreeNode root) {
        if (root == null) {
            return null;
        }
        convertBST(root.right);
        sum += root.val;
        root.val = sum;
        convertBST(root.left);
        return root;
    }
    
}
