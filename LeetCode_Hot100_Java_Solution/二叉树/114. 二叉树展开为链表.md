/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    // リストを使用してシーケンシャルトラバーサルの結果を保存し、単一のリンクリストを構築します。リスト内のノードの左側のポインターを空の値に割り当て、右側のポインターをリスト内の次の位置を指します。
   List<TreeNode> res = new ArrayList<>();
    public void flatten(TreeNode root) {
       if (root == null) {
            return;
       }
       preOrder(root);
       for (int i = 0; i < res.size() - 1; i++) {  // これはres.size () - 1未満です。配列が境界外ではないことを確認してください
        TreeNode node = res.get(i);
        node.left = null;
        node.right = res.get(i + 1);
       }
    }
    public void preOrder(TreeNode root) {
        if (root == null) {
            return;
        }
        res.add(root);
        preOrder(root.left);
        preOrder(root.right);
    }

}
