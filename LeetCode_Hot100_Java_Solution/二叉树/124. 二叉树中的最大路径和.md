/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
 // 空間複雑度の深さO (N)再帰、最悪はノード数
 // 時間複雑度O (H)木の高さ
 // 方法：再帰的+ポステリアトラバーサル
 // ノードの場合、上位ノードに提供できる最大値は、現在のノードの値+ max （左右のサブツリーの最大パス）です。
 // 現在のノードマウスの最大パスとパス内のポイントと=左右のサブツリーの最大パス+現在のノードの値
class Solution {
    private int  maxSum = Integer.MIN_VALUE;

    public int maxPathSum(TreeNode root) {
        maxGain(root);
        return maxSum;
    }

    private int maxGain(TreeNode node) {
        if (node == null) {
            return 0;
        }
        int leftGain = Math.max(maxGain(node.left), 0);
        int rightGain = Math.max(maxGain(node.right), 0);
        maxSum = Math.max(maxSum, node.val + leftGain + rightGain); // 最大値を更新
        return node.val + Math.max(leftGain, rightGain); // パスエンドポイントでない場合に貢献できる最大値を返します
    }
}
