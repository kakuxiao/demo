class Solution {
    //xとyの異なるバイナリ桁を確認して、最初にx xまたはy、x ^ yを計算することができます
    //ビット単位で再使用し、尾部が1 xor ＆ 1であるかどうかを判断します
    public int hammingDistance(int x, int y) {
        int res = x^y, hammingDistance = 0;
        while (res != 0) {
            if ((res & 1) == 1) {
                hammingDistance++;
            }
            res >>= 1;
        }
        return hammingDistance; 
    }
}
