class Solution {
    // binary
    // 現在の要素を1でビット単位で合計することによって、バイナリの1の数をカウントします
    public int[] countBits(int n) {
        int[] res= new int[n + 1];
        for (int i = 0; i <=n; i++) {
            int count = 0;
            int tmp = i;
            while (tmp != 0) {
                if ((tmp & 1) == 1) {
                    count++;
                }
                tmp >>= 1;
            }
            res[i] = count;
        }
        return res;
    }
}
