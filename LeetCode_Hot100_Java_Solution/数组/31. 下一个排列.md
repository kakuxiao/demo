class Solution {
    public void nextPermutation(int[] nums) {
        //Arrays.sort(int[] nums, int fromIndex, int toIndex) このメソッドは、numsの部分をindexからtoIndex - 1にソートします。
        // 前後に移動し、前の数値より大きい数値が表示されると、より大きな数値があることを示し、インデックスiを記録します。
        // 次の大きな数字は、次の数字を小さい行から大きい行まで押し、後続の数字の最初の数字はi -1データがそれと交換されるよりも大きい。
       int n = nums.length;
       for (int i = n - 1; i >= 0; i--) {
            if (i == 0) {
                Arrays.sort(nums);  //現在のソートは最大ソートで、最小ソートを返します
            } else if (nums[i] > nums[i-1]) {  //nums[i] > nums[i-1] 次の大きな順列があることを示します
                Arrays.sort(nums, i, n);      //そして、nums [i]の後の部分は、最大順列、次の大きな順列、すなわちnums [i]の後の部分を満たします
                for (int index = i; index < n; index++) {  // nums [i -1]より大きい最初の部分を取り出して交換します
                    if (nums[index] > nums[i-1]) {
                        int temp = nums[i - 1];
                        nums[i - 1] = nums[index];
                        nums[index] = temp;
                        return;
                    }
                }
            }
       }
 }
}
