class Solution {
    public int findDuplicate(int[] nums) {
        // 1.バイナリ検索は、要素の数以下である配列内の要素の数を計算します。数＞要素の値、説明、重複要素がl、要素間隔にある場合
        // 2.Floydサークルセンティングアルゴリズム
        // スローポインター集合場所を見つけるスローポインターで一歩ずつ、クイックポインターで一歩ずつ
        // 次に、ポインタの1つを開始点に移動します
        // 2つのポインタのミーティングポイントは、
        // リンクされたリストは、i -> nums [i]で構築することができ、重複した数字はループを示します
        int low = 0, high = nums.length -1;
        while (low < high) {
            int mid = low + (high - low) / 2;
            int count = 0;
            for (int i = 0; i < nums.length ; i++) {
                if (nums[i] <= mid) {
                    count++;
                }
            }
            if (count <= mid) {
                low = mid + 1;
            } else {
                high = mid ;
            }
        }
        return low;
