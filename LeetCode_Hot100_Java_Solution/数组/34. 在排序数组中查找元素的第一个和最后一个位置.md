class Solution {
    // 時間複雑度O (log n)空間複雑度O (1)定数空間記憶変数
    // 二分法
    // 最初のターゲットの位置を見つけ、右の枠線を更新します
    // 最後のターゲットの位置、左の境界線を更新
    public int[] searchRange(int[] nums, int target) {
        int firstPos = findBoundary(nums, target, true);
        if (firstPos == -1) {return new int[]{-1, -1};}
        int lastPos = findBoundary(nums, target, false);
        return  new int[]{firstPos, lastPos};
    }
    public int findBoundary(int[] nums, int target, boolean first) {
        int left = 0, right = nums.length - 1;
        int pos = -1;
        while (left <= right) {
            int mid = left + (right - left) / 2;
            if (nums[mid] < target) {
                left = mid + 1;
            } else if (nums[mid] > target) {
                right = mid - 1;
            } else {   // nums[mid] == target
                pos = mid;
                if (first) {
                    right = mid - 1; // ターゲットが初めて表示される場所を見つけ、右の枠線を左に移動します
                } else {
                    left = mid + 1; // ターゲットの最後の出現を見つけ、左の境界線を右に移動します
                }
            }   
        }
        return pos;
    }
    
}
