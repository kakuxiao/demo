class Solution {
    // 配列内の最初の要素で並べ替えます。
    // 前の配列の右のエンドポイントよりも小さい場合は、次の配列の左のエンドポイントをマージする必要があります
    // 2つのケース： 1.次の配列の左のエンドポイントが前の配列の右のエンドポイントよりも小さい場合は、前の配列を直接マージします
    // 2.次の配列の左エンドポイントが前の配列の右エンドポイントより大きい場合、前の配列の左エンドポイントと次の配列の右エンドポイントが取得されます
    // マージ後の要素の数がわからないため、リストを使用して最初に結果を格納する必要があります
    public int[][] merge(int[][] intervals) {
       int n = intervals.length; // 列は2として知られています
       Arrays.sort(intervals, (a,b) -> (a[0] == b[0] ? a[1] - b[1] : a[0] - b[0]));
       List<int[]> merge = new ArrayList<>();
       for (int i = 0; i < n; i++) { // intevlas[i][1]マージ中の最後の配列[1]と比較
            if (merge.isEmpty() || intervals[i][0] > merge.get(merge.size() - 1)[1]) {
                merge.add(intervals[i]);
            } else {
                merge.get(merge.size() - 1)[1] = Math.max(merge.get(merge.size() - 1)[1], intervals[i][1]);
            }
       }
    int row = merge.size();
    int[][] res = new int[row][2];
    for (int i = 0; i < row; i++) {
        for (int j = 0; j < 2; j++) {
            res[i][j] = merge.get(i)[j];
        }
    }
    return res;    
    }
}
