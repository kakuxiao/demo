''' java

    // 2. HashMapでは、時間の複雑さはO (n)であり、空間の複雑さはO (n)である。
    //合計2つの数値=ターゲットからターゲットへ- 1つの数値=もう1つのターゲット
    // one - fold for loop, iterate through the entire array, add the values that have appear to the hashmap, key is the value, value is the index
    //現在の要素hashmap.contiansKey (target - value)には2つの数値があります
    public int [] twoSum (int [] nums, int target) {
        int [] res = new int [2];
        HashMap <Integer, Integer> hs = new HashMap ();
        for (int i = 0; i &lt; nums.length; i ++) {            int value = target - nums[i];
            if (hs.containsKey(value)) {
                res[1] = i;
                res[0] = hs.get(value);
                break;
            }
            hs.put(nums[i],i);
        }
        return res;
    }
