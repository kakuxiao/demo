class Solution {
    // 質問には、o (n)の時間の複雑さが必要です
    // 質問タイプdp
    public int[] productExceptSelf(int[] nums) {
        //nums [i]を除くすべての数値の積、すなわちnums [i]は1と見なされます
        // この数値の左の積と右の積を計算できます
        // プレフィックス、サフィックス、製品をそれぞれ3つの配列で保存する
        // 要素nums [0]に対して、初期値left [0] = 1を設定します。
        // 要素nums [nums.length - 1]の初期値を右[nums.length - 1] = 1に設定します。
        int n = nums.length;
        int[] left = new int[n];
        int[] right = new int[n];
        int[] res = new int[n];
        left[0] = 1;right[n - 1] = 1;
        for (int i = 1; i < n; i++) {
            left[i] = left[i-1] * nums[i-1];
        }
        for (int j = n - 2; j >= 0; j--) {
            right[j] = right[j + 1] * nums[j + 1];
        }
        for (int i = 0; i < n; i++) {
            res[i] = left[i] * right[i];
        }
        return res;

    class Solution {
    // バックパックの問題、フルバックパック
    public int coinChange(int[] coins, int amount) {
        int[] dp = new int[amount + 1]; // dp[i] 構成金額iを表すコインの最小数
        Arrays.fill(dp, amount + 1);
        dp[0] = 0;
        for (int i = 0; i < coins.length; i++) {
            for (int j = coins[i]; j <= amount; j++) {
                dp[j] = Math.min(dp[j], dp[j-coins[i]] + 1);
            }
        }
        return dp[amount]== amount+1 ? - 1 : dp[amount];

    }
}}
}
