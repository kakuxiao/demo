class Solution {
    // o(log n) 二分法
    // 二分法の難しさは等号です
    // nums [mid]とnums [left]の関係を判断し、nums [mid] >= nums [left]の場合は、左から中が順番であることを示します
    // if (nums[mid] < nums[left]) 注文内容を中央から右に説明してください。
    // ターゲットが順序付けられた部分に該当するかどうかを判断し、該当しない場合はサブアレイを切り替えます
    // それに応じて絞り込む
    public int search(int[] nums, int target) {
        int left = 0, right = nums.length - 1;
        while (left <= right) {
            int mid = left + (right - left) / 2;
            if (nums[mid] == target) {
                return mid;
            } else if (nums[mid] >= nums[left]) { 
                if (target >= nums[left] && target < nums[mid]) {
                    right = mid - 1;
                } else {
                    left = mid + 1;
                }
            } else {
                if (target > nums[mid] && target <= nums[right]) {
                    left = mid + 1;
                } else {
                    right = mid - 1;
                }
            }
        }
        return -1;
    }
}
