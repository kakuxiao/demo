class Solution {
    //貪欲アルゴリズム、全体最適への局所最適
    //各ステップで最大値を取り、最終的にnums [nums.length - 1]に移動できるかどうかを確認します
    // if (coverrange == i) 俺が負けるまで買えないってことだ
    public boolean canJump(int[] nums) {
        int coverrange = 0;
        for (int i = 0; i < nums.length; i++) {

            coverrange = Math.max(nums[i] + i, coverrange);
            if (coverrange == i) {
                break;
            }

        }        
        return coverrange >= nums.length - 1 ? true : false;   
    }
}
