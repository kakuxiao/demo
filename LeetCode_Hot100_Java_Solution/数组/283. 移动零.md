class Solution {
    // この新しい配列の現在の要素のインデックスをポインタでマークします
    // 配列全体がトラバースされ、残りのすべてのパーツはトラバース後に0の値が割り当てられます。
    // または配列を横断するために使用される高速ポインタ、および移動後に現在のインデックスの要素を指すために使用される低速ポインタ
    public void moveZeroes(int[] nums) {
       int slow = 0;
       for (int i = 0; i < nums.length; i++) {
            if (nums[i] != 0) {
                nums[slow++] = nums[i];
            }
       }
        for (int i = slow; i < nums.length; i++) {
            nums[i] = 0;
        }
        

    }
}
