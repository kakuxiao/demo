class Solution {
    // 並べ替え場所バブル並べ替え選択並べ替えクイック並べ替え挿入並べ替え
    // 高速行平均O (nlogn)最悪O (n ^ 2)好ましくはO (nlogn)
    public void sortColors(int[] nums) {
        quickSort(nums, 0, nums.length - 1);
    }
    public void quickSort(int[] nums, int low, int high) {
        if (low < high) {
            int index = partition(nums, low ,high);
            quickSort(nums, low ,index - 1);
            quickSort(nums, index + 1,high);
        }
    }
    public int partition(int[] nums, int low, int high) { // partitionパーティション
        int pivot = nums[high];  // pivot基準
        int j = low;
        for (int i = low; i < high; i++) {
            if (nums[i] < pivot) {
                swap(nums, i, j);
                j++;
            }
        }
        swap(nums,j, high);
        return j;
    }
    public void swap(int[] arr, int i, int j) {
        int tmp = arr[i];
        arr[i] = arr[j];
        arr[j] = tmp;
    }
}
