class Solution {
    
    // ダイナミックプランニング
    // dp[i][j] (0,0)から(i, j)までの異なるパスの数を表します
    // dp[i][j] = dp[i-1][j] + dp[i][j-1];
    // 左側と上側の境界の両方に1つの可能性しかありません
    // 上から下へ、左から右へのトラバーサル
    int res = 0;
    public int uniquePaths(int m, int n) {
        int[][] dp = new int[m][n]; 
        dp[0][0] = 1;
        
        for(int i = 1; i < m;i++){
            dp[i][0] = 1;  
        }
        for(int j = 1; j < n; j++){
            dp[0][j] = 1;
        }
        for(int i = 1 ; i < m; i++){
            for(int j = 1; j < n; j++){
                dp[i][j] = dp[i - 1][j] + dp[i][j - 1]; 
            }
        } 
        return dp[m-1][n-1];
    }
   
}
