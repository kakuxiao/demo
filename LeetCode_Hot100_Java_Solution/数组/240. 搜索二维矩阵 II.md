class Solution {
    public boolean searchMatrix(int[][] matrix, int target) {
        //行列の左下または右上隅からのトラバーサル、時間複雑度O （ m + n ）
        int m = matrix.length;
        int n = matrix[0].length;
        int i = m - 1;
        int j = 0;
        while (i >= 0 && j < n) {
            if (matrix[i][j] == target) {
                return true;
            } else if(matrix[i][j] > target) {
                i--;
            } else {
                j++;
            }
        }
        
        return false;
    }

}
