class Solution {
    // 並べ替え後は、ダブルポインタ、左右ポインタを使用し、重複除外に注意してください
    // 時間複雑度O （ n ^ 2 ）、空間複雑度：複雑度O （ logn ）の並べ替え+三値リストの数O （ m ）
    public List<List<Integer>> threeSum(int[] nums) {
        List<List<Integer>> res = new ArrayList<>(); 
        Arrays.sort(nums);  // 順位
        int n = nums.length;
        for (int i = 0; i < n - 2; i++) {
            if (i > 0 && nums[i]== nums[i-1]) { //外部取り外し
                continue;
            }
            int target = -nums[i];
            int left = i + 1, right = n - 1;
            while (left < right) {
                if (nums[left] + nums[right] < target) {
                    left++;
                } else if (nums[left] + nums[right] > target) {
                    right--;
                } else {
                    res.add(Arrays.asList(nums[i], nums[left], nums[right]));
                     while (left < right && nums[right] == nums[right - 1]) { //内部重複排除
                        right--;
                    }
                    while (left < right && nums[left] == nums[left + 1]) {
                        left++;
                    }
                    left++;
                    right--;
                }
            }
        }
        return res;
    }
}
