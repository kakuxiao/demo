class Solution {
    // バックトラッキング、すべて手配済み
    // stringBufferappendおよびdeleteCharAt ()メソッド
    // 有効な括弧の組み合わせ。任意の括弧の場合、左かっこの数は右かっこの数以上です
    // 空間の複雑さ： O (n)
    List<String> res;
    int left, right;
    int count = 0;
    StringBuffer sb = new StringBuffer(); 
    public List<String> generateParenthesis(int n) {
        left = 0;
        right = 0;
        count = n;
        res = new ArrayList<>();
        backTracking();
        return res;
    }

    public void backTracking() {
        if (right == count) {
            res.add(sb.toString());
        }
        if (left < count) { // append左かっこ
            sb.append('(');
            left++;
            backTracking();
            sb.deleteCharAt(sb.length() - 1); 
            left--;
        }
        if (right < left) { // append右かっこ、右かっこの数は左かっこの数より少ない必要があります
            sb.append(')');
            right++;
            backTracking();
            sb.deleteCharAt(sb.length() - 1);
            right--;
        }
    }
}
