class Solution {
    // 使用されている数字をboolean型の配列でマークするバックトラッキングアルゴリズム
    // バックトラッキングの3つの要素：バックトラッキング関数のパラメータと戻り値の種類、バックトラッキングの終了条件、バックトラッキング検索のトラバーサルプロセス
    // フルランキング：非常に古典的なバックトラッキングアルゴリズム、バックトラッキングアルゴリズムは本質的にすべてを横断し、エネルギー効率
    List<List<Integer>> res = new ArrayList();
    List<Integer> path = new LinkedList();
    public List<List<Integer>> permute(int[] nums) {
        boolean[] used = new boolean[nums.length];
        backtracking(nums,used);
        return res;

    }
    public void backtracking(int[] nums, boolean[] used) {
        if (path.size() == nums.length) {
            res.add(new ArrayList(path));
        }
        for (int i = 0; i < nums.length; i++) {
            if (used[i] == true) continue;
            path.add(nums[i]);
            used[i] = true;
            backtracking(nums,used);
            path.remove(path.size() - 1); // 最後の2つのステップはバックトラッキングであり、現在のプレフィックス番号のすべての可能性が横断されていることを示しています
            used[i] = false; 
        }
    }
