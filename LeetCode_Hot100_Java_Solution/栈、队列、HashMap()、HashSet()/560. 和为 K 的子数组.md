
class Solution {
    //方法1 ：暴力、double forループ、最初のforループは開始を表し、2番目のforループは終了を表し、反復加算、時間の複雑さO (n ^ 2)
    //方法2 ： Prefixと+ HashMapは、2つの数値の合計と同様に、ハッシュマップを使用して時間の複雑さを軽減します
    // 
    public int subarraySum(int[] nums, int k) {
        int pre  = 0;
        int count = 0;
        HashMap <Integer,Integer> hs = new HashMap &lt;&gt; (); // hashputはプレフィックスと
        hs.put (0,1); //連続サブ配列、デフォルトは0
        for (int i = 0; i < nums.length; i++) {
            pre += nums[i];
            if (hs.containsKey(pre - k)) {
                count += hs.get(pre - k);
            }
            hs.put(pre, hs.getOrDefault(pre, 0) + 1);
        } 
        return count;
    }
}
