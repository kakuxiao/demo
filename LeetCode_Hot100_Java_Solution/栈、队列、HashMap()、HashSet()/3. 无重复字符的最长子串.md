class Solution {
    // 時間複雑性O (n )、空間複雑性O (1)
    public int lengthOfLongestSubstring(String s) {
        if (s.length() == 0) {
            return 0;
        }
       // HashSet + スライドウィンドウまたはハッシュマップ+スライドウィンドウ
       HashMap<Character, Integer> hs = new HashMap<>();
       int left = 0, right = 0, max = Integer.MIN_VALUE;
       while (right < s.length()) {
            char c = s.charAt(right);
            if (hs.containsKey(c) && hs.get(c) >= left) {  //繰り返し文字が見つかり、繰り返し文字が配置されました>左枠
                left = hs.get(c) + 1; // left 次のポジションは、最後の繰り返しポジションに1を足したものです。
            }
            hs.put(c, right);
            max = Math.max(right - left + 1, max);
            right++;
            
       }
        return max;
        
       
    }
}
