class Solution {
    public int majorityElement(int[] nums) {
        //hashmap ハッシュマップが存在する場合は値+ 1を使用し、存在しない場合は対応するキーを追加します
        Map<Integer,Integer> map = new HashMap();
        for(int num: nums){
            map.put(num,map.getOrDefault(num,0)+1);
            if(map.get(num) > nums.length /2){
                return num;
            }
        }
        return  -1;
    }
}
