class Solution {
    public boolean isValid(String s) {
        
        // ストリング内の左ブラケットは、右ブラケットをスタック内に押し込み、
        // 右の括弧が表示されると、スタックの上部にある要素と一致し、スタックの上部にある要素が対応する右の括弧であるかどうかが判定されます。
        // 不一致がある場合はFALSEを返します
        // 最終的には、スタックが空かどうかを確認する必要があります
        // 一致しない括弧を含む空の説明はありません
        
        Deque<Character> stack = new LinkedList<>();
        for (int i = 0; i < s.length(); i++) {
            if (s.charAt(i) == '(' || s.charAt(i) == '{' || s.charAt(i) == '[') {
                stack.push(s.charAt(i));
            } else {
                if (stack.isEmpty()) {
                    return false;
                }
                char top = stack.pop();
                if (s.charAt(i) == ']' && top != '[' ||  (s.charAt(i) == '}' && top != '{') || (s.charAt(i) == ')' && top != '(')) {
                    return false;
                }
            }
        }
        return stack.isEmpty(); //説明の不一致には余分な左かっこがあります

       
    }
}
