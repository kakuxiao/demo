class Solution {
    // O（n）並べ替えることはできません。
    // HashSetを使用してすべての要素を保存する
    // 最長シーケンスhashSetの現在のnums [i] - 1を含んではなりません。
    // nums [i] + 1が存在するかどうかを判断します。
    public int longestConsecutive(int[] nums) {
        if (nums.length == 0) {
            return 0;
        }
        HashSet<Integer> hs = new HashSet<>();
        for (int i = 0; i < nums.length; i++) {
            hs.add(nums[i]);
        }
        int max = 1;
       
        for (int num : nums) {
            if (hs.contains(num - 1)) {
                continue;
            }
            int maxTmp = 1;
            while (hs.contains(num + 1)) {
                num += 1;
                maxTmp++;
            }
            max = Math.max(maxTmp, max);
        }
        return max;
    }
}
