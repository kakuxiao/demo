class Solution {
    int ptr = 0;
    // スタック
    // 数字、文字列、左かっこの場合は、スタックに押し込みます
    // 右ブラケットに遭遇した場合は、隣接する左ブラケットと文字列番号をポップします
    // デコードされた文字列をスタックにプッシュします。
    // 文字列をトラバースするグローバル変数を定義する
    // StringBuffer insert (int offset, String str)は、StringBufferのinsertメソッドを使用して指定された場所に文字列を挿入するために使用されます。
    public String decodeString(String s) {
        Stack<String> stack = new Stack<>();
        while (ptr < s.length()) {
            char c = s.charAt(ptr);
            if (Character.isDigit(c)) { //数字
                String digits = getDigits(s);
                stack.push(digits);
            } else if (Character.isLetter(c) || c == '[') { 
                stack.push(String.valueOf(c));
                ptr++;
            } else {   // 右かっこ
                StringBuffer sb = new StringBuffer();
                while (!stack.peek().equals("[")) {
                    sb.insert(0, stack.pop());
                }
                String str = sb.toString(); // 
                stack.pop(); // オープニングブラケットを取り外します
                StringBuffer tmp = new StringBuffer(); 
                int repeat = Integer.parseInt(stack.pop());
                for (int j = 0; j < repeat; j++) {
                    tmp.append(str); 
                }
               stack.push(tmp.toString());
                ptr++;
            }
        }
        StringBuffer sb = new StringBuffer();
        while (!stack.isEmpty()) {
            sb.insert(0, stack.pop());
        }
        return sb.toString(); 
    }

    public String getDigits(String s) { // 番号を取得する
        StringBuffer sb = new StringBuffer();
        while (Character.isDigit(s.charAt(ptr)) && ptr < s.length()) {
            sb.append(s.charAt(ptr++));
        }
        return sb.toString();
    }



}
