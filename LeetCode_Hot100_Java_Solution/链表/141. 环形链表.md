/**
 * Definition for singly-linked list.
 * class ListNode {
 *     int val;
 *     ListNode next;
 *     ListNode(int x) {
 *         val = x;
 *         next = null;
 *     }
 * }
 */
 //ループリンクリスト、ハンズファストとスロー
public class Solution {
    // Floyd審判の輪
    public boolean hasCycle(ListNode head) {
       if (head == null || head.next == null) {
        return false;
       }
       ListNode fast = head;
       ListNode slow = head;
       while (fast != null && fast.next != null) {
            fast = fast.next.next;
            slow = slow.next;
            if (fast == slow) {
                return true;
            }
       }
       return false;
    }
}
