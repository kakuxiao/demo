/**
 * Definition for singly-linked list.
 * public class ListNode {
 *     int val;
 *     ListNode next;
 *     ListNode(int x) {
 *         val = x;
 *         next = null;
 *     }
 * }
 */
public class Solution {
    // 2つのリンクされたリストの長さを計算し、長いリストを最初にMath.abs (lenA - lenB)にします
    // その後、2つのリンクされたリストのポインタは、毎回次の位置を指します。
    // return ポインタは、同じ場所にあるノードを指します。
    public ListNode getIntersectionNode(ListNode headA, ListNode headB) {
        //リンクリストの交差ノードの代表値は等しい
        int lenA = 1, lenB = 1;
        ListNode curA = headA;
        ListNode curB = headB;
        while (curA != null) {
            curA = curA.next;
            lenA++;
        }
        while (curB != null) {
            curB = curB.next;
            lenB++;
        }
        curA = headA;
        curB = headB;
        int sub = lenA > lenB ? lenA - lenB : lenB - lenA;
        if (lenA > lenB) {
            int count = 0;
            while (count < sub) {
                curA = curA.next;
                count++;
            }
            while (curA != curB && curA != null && curB != null) {
                curA = curA.next;
                curB = curB.next; 
            }
            return curA;
        } else {
            int count = 0;
            while (count < sub) {
                curB= curB.next;
                count++;
            }
            while (curA != curB && curA != null && curB != null) {
                curA = curA.next;
                curB = curB.next; 
            }
            return curB;
        }
    }
}
