 * public class ListNode {
 *     int val;

 *     ListNode() {}
 *     ListNode(int val) { this.val = val; }

 * }

class Solution {
    // アイデア： 1.リンクリストの長さを描き、サイズ- n - 1に移動して削除します
    //      2. ポインタを遅くし、高速ポインタに最初にnステップを実行させてから、ポインタを遅くして開始します。
    //      高速ポインタが終了に達すると、遅いポインタは最後のn番目のノードの前のノードを指します
    public ListNode removeNthFromEnd(ListNode head, int n) {
          ListNode dummy = new ListNode(-1,head);
          ListNode fast = dummy;
          ListNode low = dummy;
          int count = 0;
          while (fast.next != null) { // fast最後のノードにたどり着くには、止める必要があります。
            fast = fast.next;
            count++;
            if (count >= n + 1) {
                low = low.next;
            }
          }
          low.next = low.next.next;
          return dummy.next;
