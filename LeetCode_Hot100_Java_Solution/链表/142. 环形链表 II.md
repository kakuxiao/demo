/**
 * Definition for singly-linked list.
 * class ListNode {
 *     int val;
 *     ListNode next;
 *     ListNode(int x) {
 *         val = x;
 *         next = null;
 *     }
 * }
 */
public class Solution {
    // floydサークル判定方法、クイックポインタとスローポインタ
    // 遅いを設定する両方のポインタがヘッドノードを指しています
    // 高速ポインタを一度に2歩歩かせ、遅いポインタを一歩ずつ歩かせます
    // 両手が会った後、高速ポインタを一度に1つずつ開始点に移動し、ミーティングポジションはリンクリスト内のリングの開始位置になります。
    public ListNode detectCycle(ListNode head) {
       if (head == null || head.next == null) {
            return null;
       }
       ListNode fast = head;
       ListNode slow = head;
       while (fast != null && fast.next != null) { // fast2つのステップが必要なので、素早く判断する必要があります！= null && fast.next != null
            fast = fast.next.next;
            slow = slow.next;
            if (slow == fast) {
                fast = head;
               while (fast != slow) {
                fast = fast.next;
                slow = slow.next;
               } 
               return slow;
            }
       }
        return null;
    }
}
