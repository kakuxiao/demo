class Solution {
    // 貪欲
    // 最も多くのタスクを最初に実行することで、タスク間の時間を最大限に活用できます。最も短い時間
    // 1回あたりの実行数が最も多いタスクは、プライオリティキュー、大規模なルートヒープを考慮してください
    public int leastInterval(char[] tasks, int n) {
       PriorityQueue<Integer> pq = new PriorityQueue<>((a,b) -> b-a);
       int[] count = new int[26];
       int res = 0;
       for (int i = 0; i < tasks.length; i++) { //タスクを頼りにする
            count[tasks[i] - 'A']++;
       }
       for (int tmp : count) { // 0より大きい要素の発生回数は、タスクを完了する必要があることを意味します
        if (tmp > 0) {
             pq.offer(tmp);
        }
           
       }
       while (!pq.isEmpty()) {
          List<Integer> list = new ArrayList<>(); // list一定期間内に完了したタスクを文書化する
          for (int i = 0; i <= n; i++) {
            if (!pq.isEmpty()) {
                list.add(pq.poll());
            } 
         }
         for (int t : list) {
            if ( --t > 0) {
                pq.offer(t);
            }
         }
         res += pq.isEmpty() ? list.size() : n + 1;
       } 
       return res;
    }
