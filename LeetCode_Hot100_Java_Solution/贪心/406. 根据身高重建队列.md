class Solution {
    // 短い方が並んでいて、背の高い方は見えないので、背の高い方の位置を最初に並べてから、短い方の位置を短い方の位置で並べることができます。
        // 大列から小列までの高さ（同じ高さと小さいサイズの駅前）
        //高さhで最初に降順に並べ替えます。これにより、処理時に現在挿入されている要素の前のすべての要素がその高さを下回らないようにすることができ、位置の決定が簡単になります。
        //同じ高さの場合は、昇順で数値kでソートされます。なぜなら、同じ高さの場合は、数値が最も小さい人が最初になるはずだからです。
    public int[][] reconstructQueue(int[][] people) {
        Arrays.sort(people, (a,b) -> b[0] == a[0]? a[1] - b[1] : b[0] - a[0]);
        List<int[]> list = new ArrayList<>();
        for (int i = 0; i < people.length; i++) {
            list.add(people[i][1], people[i]);
        }
       
        return list.toArray(new int[list.size()][2]); // list.toArrayリストを配列に変換
    }
}
