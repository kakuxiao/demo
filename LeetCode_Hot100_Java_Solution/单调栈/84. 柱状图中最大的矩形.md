class Solution {
    //面積、（指数の差＋ １ ） ＊ （最小高さ）
    // 単調なスタック
    // インデックス位置の最大の長方形は、高さ[index]よりも小さい左右の側面に初めて表示される長方形です。
    public int largestRectangleArea(int[] heights) {
        int n = heights.length;
        Deque<Integer> stack = new LinkedList<>();
        int maxArea = 0;
        for (int i = 0; i < heights.length; i++) {
            while (!stack.isEmpty() && heights[i] < heights[stack.peek()]) {
                int top = stack.pop(); 
                int maxWidth = stack.isEmpty() ? i : i - stack.peek() - 1; // i は右側の境界、stack.peek ()は左側の境界です。
                maxArea = Math.max(maxWidth * heights[top], maxArea);
            }
            stack.push(i);
        }
        while (!stack.isEmpty()) { // 右側の高さは小さくありません
            int top = stack.pop();
            int maxWidth = stack.isEmpty() ? n : n - stack.peek() - 1;
            maxArea = Math.max(maxWidth * heights[top], maxArea);
        }
        return maxArea;
    }
}
