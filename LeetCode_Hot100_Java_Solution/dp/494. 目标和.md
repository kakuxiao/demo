class Solution {
    // 最初に思い浮かぶのはバックトラッキングですが、バックトラッキングの時間の複雑さは大きすぎます（ 2 ^ n ）。
    // バックパックについて考える：
    // 2つのバックパック、1つはポジティブ、1つはネガティブ、
    // pos[i] - neg[i] = target;
    // pos[i] + neg[i] = sum;
    // pos[i] = (target + sum) / 2
    public int findTargetSumWays(int[] nums, int target) {
        int sum = 0;
        for (int i = 0; i < nums.length; i++) {
            sum += nums[i];
        }
        if ((target + sum) % 2 != 0 || (target + sum) < 0) {
            return 0;
        } 
        int[] pos = new int[(target + sum) / 2 + 1];
        pos[0]= 1; //pos[i] 容量のあるリュックサックを満たすことができる方法の数を示します
        for (int i = 0; i < nums.length; i++) {
            for (int j = (target + sum) / 2; j >= nums[i]; j--) {
                pos[j] += pos[j-nums[i]]; 
            }
        }
        return pos[(target + sum) / 2];
    }
}
