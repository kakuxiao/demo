class Solution {
    // dp[n] 階段のためにnに登る方法の数を表します
    // 再帰式： ｆ （ ｎ ） ＝ ｆ （ ｎ － １ ） ＋ ｆ （ ｎ － ２ ）
    // 初期値dp [0] = 1; dp [1] = 1;
    // トラバーサルの順序：小から大へ
    public int climbStairs(int n) {
        int[] res = new int[n+1];
        res[0] = 1;
        res[1] = 1;
        for(int i = 2; i <= n; i++){
            res[i] = res[i-1] + res[i-2];
        }
        return res[n];
    }
}
