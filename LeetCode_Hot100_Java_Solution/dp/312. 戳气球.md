class Solution {
    // dp[i][j] バルーンをiとjの間にスタンプすることで得られるコインの最大量を表します
    // 再帰式： dp [i ][ j ]= Math.max (dp [i ][ k ]+ dp [k ][ j ]+ val [i ]* val [j ]* val [k], dp [i ][ j])
    // k最後にポケットしたバルーンを表します
    public int maxCoins(int[] nums) {
        int n = nums.length;
        int[][] dp = new int[n + 2][n + 2];
        int[] val = new int[n + 2];
        val[0] = val[n + 1] = 1;
        for (int i = 1; i <= n; i++) {
            val[i] = nums[i - 1];
        }
        for (int len = 1; len <= n; len++) {  // つまむことができる風船の数
            for (int i = 1; i <= n - len + 1; i++) { // 突き刺すことができる出発点
                int j = i + len - 1; // 終了点
                for (int k = i; k <= j; k++) { // 最後に突き出た風船
                   dp[i][j] = Math.max (dp[i][k-1] + dp[k + 1][j] + val[i-1]*val[j+1]*val[k], dp[i][j]);
                }
            }
        }
        return dp[1][n];
    }    
}
