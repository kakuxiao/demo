class Solution {
    //ダイナミックプランニング
    //動的プログラミング直交： dp []配列とその下付き文字、再帰式、初期化、トラバーサル順序の意味を決定する
    // 3つの状態：在庫を保持し、在庫を保持せず、凍結期間にあり、在庫を保持せず、dp [i ][ 0 ]、dp [i ][ 1 ]、dp [i ][ 2]に対応する凍結期間にありません。
    //  再帰式： dp [i ][ 0] = dp [i -1 ][ 2 ]- prices [0];
    // dp[i][1] = dp[i][0] + prices[i] dp[i][2] = dp[i-1][1];
    public int maxProfit(int[] prices) {
       int n = prices.length;
       int[][] dp = new int[n][3];
       dp[0][0] = -prices[0];
       dp[0][1] = 0; dp[0][2] = 0;
       for (int i = 1; i < prices.length; i++) {
        dp[i][0] = Math.max(dp[i-1][0], dp[i-1][2] - prices[i]);
        dp[i][1] = dp[i-1][0] + prices[i];
        dp[i][2] = Math.max(dp[i-1][2], dp[i-1][1]);
       }
       return Math.max(dp[n-1][1],dp[n-1][2]);
        
    }
}
