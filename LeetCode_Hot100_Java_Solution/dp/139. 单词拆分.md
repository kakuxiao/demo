class Solution {
    //動的プログラミングdp []は、最初のi文字がリスト内の文字列で構成できるかどうかを示します
    //初期化dp [0] = 0;
    //トラバーサルの順序：左から右
    //再帰数式
    //s.substring(start,endIndex) startは開始インデックス、endIndexは終了インデックスです。
    // 結合の代わりに配置する
    public boolean wordBreak(String s, List<String> wordDict) {
            int n = s.length(); 
            boolean[] dp = new boolean[s.length() + 1];  //sの最初のi個の要素が辞書の要素で構成できるかどうかを表します
            dp[0] = true;
            for (int i = 1; i <= n; i++) {
                for (String str : wordDict) {
                    if (i >= str.length() && dp [i - str.length()] && s.substring(i - str.length(), i).equals(str)) {
                        dp[i] = true;
                    }
                }
            } 
            return dp[n];
    }
}
