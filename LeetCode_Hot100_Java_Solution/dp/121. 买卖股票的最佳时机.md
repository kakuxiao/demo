class Solution {
    // 要素の右側にある要素の最大値を見つけ、それを悪化させ、サイズが悪化し、暴力的なメソッドがタイムアウトします
    // 特定の日に、その日に株式を売却することによって得られる最大値は、現在の株価から以前に発生した最小値を差し引いた値です。
    // プロセス全体の最大値を見つける
    public int maxProfit(int[] prices) {
        int max = 0;
        int min = Integer.MAX_VALUE;
        for (int i = 0; i < prices.length; i++) {
            if (prices[i]  < min) {
                min = prices[i];
            }
            if (prices[i] - min > max) {
                max = prices[i] - min;
                
            }
        }
        return max;
    }
}
