class Solution {
    //動的プログラミング、レビューdp [i] [j]は、正規表現マッチングを達成するために、文字列sの最初のi文字を文字列pの最初のjの文字列とマッチングできるかどうかを示します
  public boolean isMatch(String s, String p) {
       int m = s.length();
       int n = p.length();
       boolean[][] dp = new boolean[m+1][n+1];
       dp[0][0] = true;
       for (int i = 1; i <= n; i++) {
            if (p.charAt(i-1) == '*') {
                dp[0][i] = dp[0][i-2];
            }
       }
       for (int i = 1; i <= m; i++) {
        for (int j = 1; j <= n; j++) {
            if (p.charAt(j - 1) == '.' || p.charAt(j-1) == s.charAt(i-1)) {
                dp[i][j] = dp[i-1][j-1];
            } else if (p.charAt(j-1) == '*') {
                dp[i][j] = dp[i][j-2] || ((s.charAt(i-1) == p.charAt(j-2) || p.charAt(j-2) == '.')&&dp[i-1][j] );
                // 0試合、複数試合
            }
        }
       }
       return dp[m][n];
    }
}
