        int n = s.length();
        if (n == 1) {
            return s;
        }
        boolean[][] dp = new boolean[n][n];
        for (int i = 0; i < n; i++) {
            dp[i][i] = true;
        }
        int maxLen = 1;
        int begin = 0;
        for (int i = 2; i <= n; i++) {
            for (int j = 0; j < n;j++) {
                int k = j + i - 1;
                if(k >= n) {
                    break;
                }
                if (i <= 3) {
                    if (s.charAt(j) == s.charAt(k)) {
                        dp[j][k] = true;
                    }else {
                        dp[j][k] = false;
                    }
                } else {
                     if (s.charAt(j) == s.charAt(k)) {
                        dp[j][k] = dp[j +1][k - 1] ;
                    }else {
                        dp[j][k] = false;
                    }
                }
                if (dp[j][k] && i > maxLen ) {
                    maxLen = i;
                    begin = j;
                }
            }
        }
        return s.substring(begin,begin+maxLen);

    }
