class Solution {
    // 現在の要素を行列の右下隅として使用し、現在の行の連続する1の数を配列で現在の位置に記録して、行列の最大幅を取得します
    // 垂直に横断し、幅と高さを計算し、面積を取得し、対応する位置領域の最大値を求める
    public int maximalRectangle(char[][] matrix) {
        int row = matrix.length;
        int col = matrix[0].length;
        int maxArea = 0;
        int[][] maxWidth = new int[row][col]; // 現在の要素の右下隅に最大幅を記録します
        for (int i = 0; i < row; i++) {
            for (int j = 0; j < col; j++) {
                if (matrix[i][j] == '1') {
                    maxWidth[i][j] = (j > 0) ? maxWidth[i][j - 1] + 1 : 1;
                }
            }
        }
        for (int i = 0; i < row; i++) {
            for (int j = 0; j < col; j++) {
                if (matrix[i][j] == '1') {
                    int width = maxWidth[i][j];
                    int area = width;
                    for (int k = i - 1; k >= 0; k--) {
                        width = Math.min(maxWidth[k][j], width); //長方形の最大幅を構築
                        area = Math.max(width*(i - k + 1), area);
                    }
                    maxArea = Math.max(area, maxArea);
                }
            }
        }
        return maxArea;
    }
}
