class Solution {
    // ダイナミックプランニング
    // dp[i] iで終わる最も長い有効な括弧の部分文字列の長さを表します
    // s.charAt(i) == '( dp[i] = 0;
    // s.charAt(i) == ')' s.charAt(i - 1) = '(' dp[i] = dp[i - 2] + 2
    // if (i - dp[i - 1] - 2 >= 0)s.charAt(i - dp[i - 1] - 1) == '(' dp[i] = dp[i-1] + 2 + dp[i - dp[i - 1] - 2]
    // 初期化dp [0] = 0;
    // 左から右へのトラバーサル順序
    public int longestValidParentheses(String s) {
        int len = s.length();
        int[] dp = new int[len];
        int maxLen = 0;
        for (int i = 1; i <= len - 1; i++) {
            if (s.charAt(i) == ')') {
                if (s.charAt(i-1) == '(') {
                    dp[i] = (i >= 2 ? dp[i-2] : 0) + 2;
                    maxLen = Math.max(maxLen,dp[i]);
                } else if (i-1 >= dp[i-1] && dp[i-1] > 0 && s.charAt(i-dp[i-1]-1) == '(') { // dp [i -1] > 0を判断する必要がある
                    dp[i] = (i - dp[i-1] >= 2)? dp[i-1] + dp[i-dp[i-1]-2] + 2 : dp[i-1] + 2;
                }
                 maxLen = Math.max(dp[i],maxLen);
            }
        }
        return maxLen;
    }
}
