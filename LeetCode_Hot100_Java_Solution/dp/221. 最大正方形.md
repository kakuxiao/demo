class Solution {
    // 動的プログラミング： dp [i ][ j]は、右下隅にnums [i ][ j]を持つ正方形の最大値を表します
    // 再帰式： dp [i ][ j ]= Math.min (dp [i -1 ][ j], dp [i ][ j - 1], dp [i - 1 ][ j - 1])
    // 初期化i == 0 || j == 0の場合を初期化します
    // 左から右へ、上から下へのトラバーサル順序
    public int maximalSquare(char[][] matrix) {
        int m = matrix.length;
        int n = matrix[0].length;
        int[][] dp = new int[m][n];
        int maxSide = 0;
        for (int i = 0; i < m; i++) {
            if (matrix[i][0] == '1') {
                dp[i][0] = 1;
                maxSide = 1; 
            }
        }
        for (int j = 0; j < n; j++) {
            if (matrix[0][j] == '1') {
                dp[0][j] = 1;
                maxSide = 1;
                
            }
        }
        for (int i = 1; i < m; i++) {
            for (int j = 1; j < n; j++) {
                if (matrix[i][j] == '1') {
                    dp[i][j] = Math.min(dp[i - 1][j],Math.min(dp[i][j - 1], dp[i - 1][j - 1])) + 1;
                    maxSide = Math.max(maxSide, dp[i][j]);
                }
            }
        }
        return maxSide*maxSide;
    }
}
