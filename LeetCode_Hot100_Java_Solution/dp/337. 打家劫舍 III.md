/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    //ノードごとに、選択するか選択しないかの2つの決定があります。
    //このノードが選択されている場合、その左の子ノードと右の子ノードは選択できません
    //配列を設定します。下付き文字0はノードに選択された最大盗難金額を表し、下付き文字1は選択されていないノードの最小盗難金額を表します。
    // dfs + dp
    // 時間の複雑さO (n )、空間の複雑さO (n);各ノードは1回通過します
    public int rob(TreeNode root) {
        int[] res = dfs(root); // res[]ルートを選択し、ルートの選択を解除して、盗まれる可能性のある最大値を記録します
        return Math.max(res[0], res[1]);
    }
    public int[] dfs(TreeNode root) {
        if (root == null) {
            return new int[]{0,0};
        }
        int[] left = dfs(root.left);
        int[] right = dfs(root.right);
        int selected = root.val + left[1] + right[1];
        int noSelected = Math.max(left[0], left[1]) + Math.max (right[0], right[1]);
        return new int[]{selected, noSelected};
    }
    
}
