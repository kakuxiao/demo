class Solution {
    //ダイナミックプランニングの4つのステップ: 1. dp配列と下付き文字の意味を決定する
    // 2.再帰数式を決定する
    // 3.dp配列の初期化方法
    // 4.トラバーサルの順序を決定する
    // 現在の家を盗むのではなく、現在の家を盗む
    // dp[i] 部屋0 - iの盗難から得られる最大値を表します
    // dp[i] = Math.max(dp[i-1], dp[i-2] +nums[i]);
    // dp[0] = nums[0], dp[1] = Math.max(nums[1], nums[0]);
    //
    public int rob(int[] nums) {
        int n = nums.length;
        if (n == 1) {
            return nums[0];
        } else if (n == 2) {
            return Math.max(nums[0], nums[1]);
        }
        int[] dp = new int[n];
        dp[0] = nums[0]; dp[1] = Math.max(nums[0], nums[1]);
        for (int i = 2; i < n; i++) {
            dp[i] = Math.max(dp[i-1], dp[i - 2] + nums[i]); //次の部屋への2つの可能性のそれぞれについて、1.現在の部屋を盗む。2.現在の部屋を盗まないでください
        }
        return dp[n-1];
    }
}
