        int n = nums.length;
        int[] dpMax = new int[n];
        int[] dpMin = new int[n];
        int res = nums[0];
        dpMax[0] = nums[0];dpMin[0] = nums[0];
        for (int i = 1; i < n; i++) {
            dpMax[i] = Math.max(nums[i],Math.max(nums[i]*dpMax[i-1],nums[i]*dpMin[i-1]));
            dpMin[i] = Math.min(nums[i], Math.min(nums[i] * dpMax[i-1],nums[i]*dpMin[i-1]));
            res = Math.max(res, dpMax[i]);
        }
        return res;
