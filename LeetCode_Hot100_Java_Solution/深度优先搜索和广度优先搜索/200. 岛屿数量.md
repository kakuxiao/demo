class Solution {
    // 深度優先検索：一方向に反復し、最初に戻り、反対方向に反復します
    // 幅優先検索： 1つの位置から開始し、重複を避けるために、キューで実装されたレイヤーごとに隣接する位置まで検索し、ブール配列でマークする必要があります


     // 島の数に関する質問は、深さ優先探索と幅優先探索の基礎となります
     // 深度優先コードは比較的単純で、文字配列を横断し、プロセスは'1'に遭遇してディープサーチを行います。ディープサーチのプロセスが境界外であるかどうかを判断する限り、要素は'1'であり、深度優先は隣接する'1'に与えられます。
    int[][] pos = {{1,0}, {-1,0},{0,1},{0,-1}};
    public int numIslands(char[][] grid) {
        int m = grid.length;
        int n = grid[0].length;
        int num = 0;
        boolean[][] visited = new boolean[m][n];
        for (int i = 0; i < grid.length; i++) {
            for (int j = 0; j < grid[0].length; j++) {
                if ( grid[i][j] == '1') {
                    dfs(grid,i,j);
                    num++;
                }
            }
        }
        return num;
    }
    public void dfs(char[][] grid, int row, int col) {
        if (row < 0 || row >= grid.length || col < 0 || col >= grid[0].length || grid[row][col] == '0') {
             return;
         }
        
        grid[row][col] = '0';
        dfs(grid, row + 1, col);
        dfs(grid, row - 1, col);
        dfs(grid, row, col + 1);
        dfs(grid, row, col - 1);
    }

    public void bfs(char[][] grid, int row, int col, boolean[][] visited) {
        grid[row][col] = '0';
        visited[row][col] = true;
        Queue<int[]> que = new LinkedList<>();
        que.offer(new int[]{row,col});
        while (!que.isEmpty()) {
            int[] cur = que.poll();
            int x = cur[0];
            int y = cur[1];
            for (int[] tmp : pos) {
                int curX = tmp[0] + x;
                int curY = tmp[1] + y;
                if (curX < 0 || curX >= grid.length || curY < 0 || curY >= grid[0].length || grid[curX][curY] == '0' || visited[curX][curY]) {
                    continue;
                }
                que.offer(new int[]{curX, curY});
                visited[curX][curY] = true;
            }
            
        }
    }
}
